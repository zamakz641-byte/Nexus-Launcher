cmake_minimum_required(VERSION 3.24)
project(NexusReplay VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS src/capture/*.cpp src/capture/*.h src/capture/hdr/*.cpp src/capture/hdr/*.h src/core/*.cpp)
add_executable(NexusReplay src/main.cpp ${CORE_SOURCES})
target_include_directories(NexusReplay PRIVATE src)
target_compile_definitions(NexusReplay PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(NexusReplay PRIVATE Qt6::Core Qt6::Gui user32 gdi32 windowsapp runtimeobject ole32 oleaut32 d3d11 dxgi dxguid uuid d3dcompiler mfplat mfreadwrite mfuuid mf avrt ksuser mmdevapi)
if(MINGW)
  target_link_options(NexusReplay PRIVATE -Wl,--gc-sections)
  target_compile_options(NexusReplay PRIVATE -ffunction-sections -fdata-sections)
endif()
